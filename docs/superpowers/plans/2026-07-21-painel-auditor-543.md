```sql
-- 0019_eval_cycles.sql
-- Introduz o conceito de ciclo (janela temporal) para o motor eval_*.
-- Só um ciclo pode estar 'active=true' por vez (partial unique index).
-- Backfill: instâncias antigas caem no ciclo 2026 (o ativo pós-migration).

CREATE TABLE IF NOT EXISTS eval_cycles (
  id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  label       text NOT NULL,
  start_date  date NOT NULL,
  end_date    date NOT NULL,
  active      boolean NOT NULL DEFAULT false,
  created_at  timestamptz NOT NULL DEFAULT now(),
  CHECK (end_date >= start_date)
);
CREATE UNIQUE INDEX IF NOT EXISTS eval_cycles_one_active
  ON eval_cycles (active) WHERE active;

ALTER TABLE eval_instances
  ADD COLUMN IF NOT EXISTS cycle_id bigint REFERENCES eval_cycles(id) ON DELETE SET NULL;
CREATE INDEX IF NOT EXISTS eval_instances_cycle_idx
  ON eval_instances (cycle_id);

-- Seed idempotente: só insere se ainda não houver nenhum ciclo.
INSERT INTO eval_cycles (label, start_date, end_date, active)
SELECT 'Ciclo 2026', DATE '2026-01-01', DATE '2026-12-31', true
WHERE NOT EXISTS (SELECT 1 FROM eval_cycles);

-- Backfill: qualquer instância órfã vai para o único ciclo existente.
UPDATE eval_instances SET cycle_id = (SELECT id FROM eval_cycles ORDER BY id LIMIT 1)
 WHERE cycle_id IS NULL;
```

```typescript
  // ==========================================================================
  // Ciclos de qualificação (eval_cycles) — Painel do Auditor (spec 5.4.2.5)
  // ==========================================================================

  const isStaff = (role: string) => role === 'admin' || role === 'auditoria';

  app.get('/api/eval/cycles', authenticate, async (req: any, res) => {
    try {
      if (!isStaff(req.user.role)) return res.status(403).json({ error: 'Unauthorized' });
      const { rows } = await query(`
        SELECT c.id, c.label, c.start_date, c.end_date, c.active,
               (SELECT COUNT(*)::int FROM eval_instances ei WHERE ei.cycle_id = c.id) AS instances_count
          FROM eval_cycles c
         ORDER BY c.active DESC, c.start_date DESC, c.id DESC
      `);
      res.json(rows);
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: err.message });
    }
  });

  app.post('/api/eval/cycles', authenticate, async (req: any, res) => {
    try {
      if (!isStaff(req.user.role)) return res.status(403).json({ error: 'Unauthorized' });
      const { label, start_date, end_date, active } = req.body || {};
      if (!label || !start_date || !end_date) {
        return res.status(400).json({ error: 'label, start_date, end_date obrigatórios' });
      }
      if (new Date(end_date) < new Date(start_date)) {
        return res.status(400).json({ error: 'end_date < start_date' });
      }
      await query('BEGIN');
      try {
        if (active) {
          await query('UPDATE eval_cycles SET active = false WHERE active = true');
        }
        const { rows } = await query(
          `INSERT INTO eval_cycles (label, start_date, end_date, active)
           VALUES ($1, $2, $3, COALESCE($4, false)) RETURNING *`,
          [label, start_date, end_date, active || false]
        );
        await query('COMMIT');
        res.json(rows[0]);
      } catch (e) {
        await query('ROLLBACK');
        throw e;
      }
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: err.message });
    }
  });

  app.patch('/api/eval/cycles/:id', authenticate, async (req: any, res) => {
    try {
      if (!isStaff(req.user.role)) return res.status(403).json({ error: 'Unauthorized' });
      const id = Number(req.params.id);
      const { label, start_date, end_date, active } = req.body || {};
      await query('BEGIN');
      try {
        if (active === true) {
          await query('UPDATE eval_cycles SET active = false WHERE active = true AND id <> $1', [id]);
        }
        const { rows } = await query(
          `UPDATE eval_cycles
              SET label      = COALESCE($1, label),
                  start_date = COALESCE($2, start_date),
                  end_date   = COALESCE($3, end_date),
                  active     = COALESCE($4, active)
            WHERE id = $5 RETURNING *`,
          [label ?? null, start_date ?? null, end_date ?? null, active ?? null, id]
        );
        if (!rows.length) { await query('ROLLBACK'); return res.status(404).json({ error: 'not found' }); }
        await query('COMMIT');
        res.json(rows[0]);
      } catch (e) {
        await query('ROLLBACK');
        throw e;
      }
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: err.message });
    }
  });

  app.delete('/api/eval/cycles/:id', authenticate, async (req: any, res) => {
    try {
      if (!isStaff(req.user.role)) return res.status(403).json({ error: 'Unauthorized' });
      const id = Number(req.params.id);
      const { rows: refs } = await query(
        'SELECT 1 FROM eval_instances WHERE cycle_id = $1 LIMIT 1', [id]
      );
      if (refs.length) return res.status(409).json({ error: 'ciclo tem instâncias vinculadas' });
      await query('DELETE FROM eval_cycles WHERE id = $1', [id]);
      res.status(204).end();
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: err.message });
    }
  });
```

Arquivo `src/lib/auditorDashboard.ts`:

```typescript
export interface DashboardInstance {
  id: number;
  provider_id: number;
  location_id: number | null;
  provider_name: string;
  provider_code: string | null;
  city: string | null;
  provider_group: string | null;
  status: string;
  percentual: number;
  seal: string | null;
  has_certificate: boolean;
}

export interface DimPct { instance_id: number; dimension: string; pct: number; }

export interface AuditorDashboardPayload {
  cycle: { id: number; label: string; start_date: string; end_date: string };
  kpis: {
    total_locations: number;
    sent: number; pending: number;
    in_review: number; awaiting_provider: number;
    avg_score: number;
    low_security_count: number;
    seal_distribution: Record<string, number>;
    group_distribution: { group: string; count: number }[];
  };
  instances: DashboardInstance[];
}

const SENT_STATUSES = new Set(['em_analise', 'aguardando_evidencia', 'aprovado', 'reprovado', 'finalizado']);
const PENDING_STATUSES = new Set(['em_andamento', 'pendente']);

export function buildAuditorDashboard(
  cycle: AuditorDashboardPayload['cycle'],
  instances: DashboardInstance[],
  dimPct: DimPct[],
): AuditorDashboardPayload {
  const pairKey = (i: DashboardInstance) => `${i.provider_id}::${i.location_id ?? 'null'}`;
  const distinctPairs = new Set(instances.map(pairKey));
  const total_locations = distinctPairs.size;

  let sent = 0, pending = 0, in_review = 0, awaiting_provider = 0;
  const scores: number[] = [];
  const seal_distribution: Record<string, number> = { 'Ouro': 0, 'Prata': 0, 'Bronze': 0, 'Não Qualificado': 0 };
  const groupPairs = new Map<string, Set<string>>();

  for (const inst of instances) {
    if (SENT_STATUSES.has(inst.status)) sent++;
    if (PENDING_STATUSES.has(inst.status)) pending++;
    if (inst.status === 'em_analise') in_review++;
    if (inst.status === 'aguardando_evidencia') awaiting_provider++;
    if (inst.status !== 'em_andamento') scores.push(Number(inst.percentual) || 0);

    if (inst.seal && seal_distribution[inst.seal] !== undefined) seal_distribution[inst.seal]++;
    else if (!inst.seal && inst.status === 'reprovado') seal_distribution['Não Qualificado']++;

    if (inst.provider_group) {
      if (!groupPairs.has(inst.provider_group)) groupPairs.set(inst.provider_group, new Set());
      groupPairs.get(inst.provider_group)!.add(pairKey(inst));
    }
  }

  const avg_score = scores.length ? Math.round((scores.reduce((s, x) => s + x, 0) / scores.length) * 10) / 10 : 0;

  const secByInstance = new Map<number, number>();
  for (const d of dimPct) {
    if (d.dimension === 'Segurança') secByInstance.set(d.instance_id, d.pct);
  }
  let low_security_count = 0;
  for (const pct of secByInstance.values()) if (pct < 60) low_security_count++;

  const group_distribution = Array.from(groupPairs.entries())
```

```typescript
import XLSX from 'xlsx';

export interface XlsxSourceRow {
  provider_name: string;
  provider_code: string | null;
  city: string | null;
  address: string | null;
  percentual: number;
  seal: string | null;
  started_at: string | null;
  completed_at: string | null;
  status: string;
}

export const AUDITOR_XLSX_HEADERS = [
  'Município', 'Logradouro', 'Prestador', 'Código',
  'Pontuação Total', 'Classificação',
  '% Segurança', '% Conforto', '% Experiência do Cliente',
  'Data de Envio', 'Data de Conclusão da Análise',
] as const;

const FINAL_STATUSES = new Set(['aprovado', 'reprovado', 'finalizado']);

export function buildAuditorXlsxRows(
  instances: (XlsxSourceRow & { id: number })[],
  dimPctByInstance: Record<number, Record<string, number>>,
): (string | number | null)[][] {
  const rows: (string | number | null)[][] = [[...AUDITOR_XLSX_HEADERS]];
  for (const i of instances) {
    const dims = dimPctByInstance[i.id] || {};
    const pct = (name: string) => (dims[name] != null ? dims[name] : null);
    rows.push([
      i.city ?? null,
      i.address ?? null,
      i.provider_name,
      i.provider_code ?? null,
      Number(i.percentual) || 0,
      i.seal ?? null,
      pct('Segurança'),
      pct('Conforto'),
      pct('Experiência do Cliente'),
      i.started_at ?? null,
      FINAL_STATUSES.has(i.status) ? (i.completed_at ?? null) : null,
    ]);
  }
  return rows;
}

export function serializeAuditorXlsx(rows: (string | number | null)[][]): Buffer {
  const ws = XLSX.utils.aoa_to_sheet(rows);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, 'Painel do Ciclo');
  return XLSX.write(wb, { type: 'buffer', bookType: 'xlsx' }) as Buffer;
}
```

```tsx
import { fetchApi } from '../../lib/api';

export default function NovoCicloModal({ initial, onClose, onSaved }: { initial?: any; onClose: () => void; onSaved: () => void }) {

  const [start, setStart] = useState(initial?.start_date?.slice(0, 10) || '');
```

```tsx
              <th className="p-3 text-sm font-semibold text-gray-600">Ciclo</th>
              <th className="p-3 text-sm font-semibold text-gray-600">Início</th>
              <th className="p-3 text-sm font-semibold text-gray-600">Fim</th>
              <th className="p-3 text-sm font-semibold text-gray-600">Ativo</th>
              <th className="p-3 text-sm font-semibold text-gray-600">Instâncias</th>
              <th className="p-3 text-sm font-semibold text-gray-600">Ações</th>
            </tr>
          </thead>
          <tbody>
            {loading ? (
              <tr><td colSpan={6} className="p-6 text-center text-gray-500">Carregando...</td></tr>
            ) : cycles.length === 0 ? (
              <tr><td colSpan={6} className="p-6 text-center text-gray-500">Nenhum ciclo cadastrado.</td></tr>
            ) : cycles.map(c => (
              <tr key={c.id} className="border-b border-gray-50">
                <td className="p-3 font-medium">{c.label}</td>
                <td className="p-3 text-sm text-gray-600">{String(c.start_date).slice(0, 10)}</td>
                <td className="p-3 text-sm text-gray-600">{String(c.end_date).slice(0, 10)}</td>
                <td className="p-3">
                  {c.active
                    ? <span className="px-2 py-0.5 text-xs rounded bg-emerald-100 text-emerald-700 font-semibold">Ativo</span>
                    : <button onClick={() => activate(c)} className="text-xs text-emerald-700 hover:underline flex items-center gap-1"><Check className="w-3 h-3" /> Ativar</button>}
                </td>
                <td className="p-3 text-sm text-gray-600">{c.instances_count}</td>
                <td className="p-3">
                  <div className="flex gap-2">
                    <button onClick={() => setEditing(c)} className="text-blue-600 hover:text-blue-800" title="Editar"><Edit3 className="w-4 h-4" /></button>
                    <button
                      onClick={() => remove(c)}
                      disabled={c.instances_count > 0}
                      title={c.instances_count > 0 ? `Não pode excluir — ${c.instances_count} instâncias vinculadas` : 'Excluir'}
                      className="text-red-600 hover:text-red-800 disabled:opacity-30 disabled:cursor-not-allowed"
                    ><Trash2 className="w-4 h-4" /></button>
                  </div>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {creating && <NovoCicloModal onClose={() => setCreating(false)} onSaved={load} />}
      {editing && <NovoCicloModal initial={editing} onClose={() => setEditing(null)} onSaved={load} />}
    </div>
  );
}
```
