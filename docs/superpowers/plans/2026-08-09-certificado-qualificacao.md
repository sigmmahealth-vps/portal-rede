# Certificado de Qualificação (5.4.5) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Emitir certificado digital automaticamente no parecer aprovado, com aba "Meus Certificados" para o prestador, PDF via página de impressão e verificação pública por QR code.

**Architecture:** Tabela mínima `eval_certificates` (verify_code aleatório + validade) preenchida na transação do `POST /parecer` e por backfill na migration. Dados exibidos vêm por JOIN de `eval_instances`/`providers`/`provider_locations`/`eval_cycles`. Frontend: nova aba no `PrestadorView`, página de impressão A4 paisagem (window.print) e página pública `/verificar/:codigo`.

**Tech Stack:** Express + pg (server.ts monolítico), React 19 + react-router 7 + Tailwind 4, `node --test`, date-fns, dependência nova: `qrcode` (+ `@types/qrcode` dev).

**Spec:** `docs/superpowers/specs/2026-08-09-certificado-qualificacao-design.md`

**Convenções do projeto que valem para TODAS as tasks:**
- Working dir: `portal-rede-v1-main/` (todos os paths abaixo são relativos a ela).
- Endpoints do prestador logado usam `req.user.provider_id` (403 se ausente) — padrão de `GET /api/me/action-plans` (server.ts:4685). Por isso o endpoint de lista é `GET /api/me/certificates` (convenção `/api/me/...`), não `/api/eval/certificados` como o spec esboçou.

```sql
-- 0025_eval_certificates.sql
-- Certificado de Qualificação (spec 5.4.5). Um certificado por instância
-- aprovada; verify_code é o código público de autenticidade (QR).
-- Validade = end_date do ciclo + 1 ano (fallback: completed_at + 1 ano).

CREATE TABLE IF NOT EXISTS eval_certificates (
  id           bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  instance_id  bigint NOT NULL UNIQUE REFERENCES eval_instances(id) ON DELETE CASCADE,
  verify_code  text   NOT NULL UNIQUE,   -- 32 chars hex aleatórios
  issued_at    timestamptz NOT NULL DEFAULT now(),
  valid_until  date   NOT NULL,
  created_at   timestamptz NOT NULL DEFAULT now()
);

-- Backfill: instâncias já aprovadas antes da feature ganham certificado
-- retroativo com issued_at = completed_at (data honesta da aprovação).
INSERT INTO eval_certificates (instance_id, verify_code, issued_at, valid_until)
SELECT ei.id,
       replace(gen_random_uuid()::text, '-', ''),
       COALESCE(ei.completed_at, now()),
       (COALESCE(c.end_date, COALESCE(ei.completed_at, now())::date) + interval '1 year')::date
  FROM eval_instances ei
  LEFT JOIN eval_cycles c ON c.id = ei.cycle_id
 WHERE ei.status IN ('aprovado', 'finalizado')
ON CONFLICT (instance_id) DO NOTHING;
```

```ts
// src/lib/certValidity.ts
import { addYears, format, parseISO } from 'date-fns';

/**
 * Validade do certificado (spec 5.4.5): end_date do ciclo + 1 ano.
 * Fallback (instância sem ciclo): data de aprovação + 1 ano.
 *
 * cycleEndDate pode chegar como 'YYYY-MM-DD' (SQL) ou Date (driver pg).
 * String date-only DEVE ser parseada com parseISO (meia-noite local) —
 * `new Date('2026-12-31')` seria UTC e regrediria um dia em UTC-3.
 */
export function computeValidUntil(
  cycleEndDate: string | Date | null,
  fallback: Date,
): string {
  const base = cycleEndDate == null
    ? fallback
    : typeof cycleEndDate === 'string' ? parseISO(cycleEndDate) : cycleEndDate;
  return format(addYears(base, 1), 'yyyy-MM-dd');
}
```

```ts
  // Meus certificados (spec 5.4.5 — visão do prestador). Um por instância
  // aprovada; o seletor de ciclo é client-side (payload traz cycle_id/label).
  app.get('/api/me/certificates', authenticate, async (req: any, res) => {
    try {
      const providerId = req.user.provider_id;
      if (!providerId) return res.status(403).json({ error: 'Unauthorized' });

      const { rows: certs } = await query(`
        SELECT ec.id, ec.verify_code, ec.issued_at, ec.valid_until,
               ei.id AS instance_id, ei.seal, ei.percentual,
               c.id AS cycle_id, c.label AS cycle_label, c.active AS cycle_active,
               COALESCE(p.trade_name, p.name) AS provider_name,
               pl.label AS location_label,
               pl.logradouro, pl.numero, pl.bairro, pl.cidade, pl.uf
          FROM eval_certificates ec
          JOIN eval_instances ei ON ei.id = ec.instance_id
          JOIN providers p ON p.id = ei.provider_id
          LEFT JOIN provider_locations pl ON pl.id = ei.location_id
          LEFT JOIN eval_cycles c ON c.id = ei.cycle_id
         WHERE ei.provider_id = $1
         ORDER BY ec.issued_at DESC, ec.id DESC
      `, [providerId]);

      // Desempenho por dimensão (mesma consolidação DL→Segurança do painel).
      const ids = certs.map((r: any) => Number(r.instance_id));
      const dimByInstance = new Map<number, { dimension: string; pct: number }[]>();
      if (ids.length) {
        const { rows: dims } = await query(`
          SELECT ea.instance_id,
                 CASE WHEN eq.dimension IN ('Segurança','Documento Legal') THEN 'Segurança' ELSE eq.dimension END AS dimension,
                 (SUM(ea.score_obtained) / NULLIF(SUM(eq.max_score), 0)) * 100 AS pct
            FROM eval_answers ea
            JOIN eval_questions eq ON eq.id = ea.question_id
           WHERE ea.instance_id = ANY($1)
             AND LOWER(COALESCE(ea.answer_value,'')) <> 'na'
             AND eq.dimension IS NOT NULL
           GROUP BY ea.instance_id,
                    CASE WHEN eq.dimension IN ('Segurança','Documento Legal') THEN 'Segurança' ELSE eq.dimension END
           ORDER BY 2
        `, [ids]);
        for (const d of dims) {
          const k = Number(d.instance_id);
          if (!dimByInstance.has(k)) dimByInstance.set(k, []);
          dimByInstance.get(k)!.push({ dimension: d.dimension, pct: Math.round(Number(d.pct) * 10) / 10 });
        }
      }

      res.json(certs.map((r: any) => {
        const endereco = [
          [r.logradouro, r.numero].filter(Boolean).join(', '),
          r.bairro, r.cidade,
        ].filter(Boolean).join(', ') || null;
        return {
          id: Number(r.id),
          verify_code: r.verify_code,
          issued_at: r.issued_at,
          valid_until: r.valid_until,
          seal: r.seal,
          percentual: Number(r.percentual) || 0,
          cycle_id: r.cycle_id != null ? Number(r.cycle_id) : null,
          cycle_label: r.cycle_label,
          cycle_active: !!r.cycle_active,
          provider_name: r.provider_name,
          location_label: r.location_label,
          location_address: endereco,
          dimensions: dimByInstance.get(Number(r.instance_id)) ?? [],
        };
      }));
    } catch (err: any) { console.error(err); res.status(500).json({ error: err.message }); }
  });
```

### Task 6: `GET /api/certificados/verificar/:code` — verificação pública

**Files:**
- Modify: `server.ts` — inserir logo após o `GET /api/me/certificates` da Task 5

- [ ] **Step 1: Implementar o endpoint (SEM authenticate)**

```ts
  // Verificação pública de autenticidade (spec 5.4.5) — alvo do QR code.
  // Sem login. Rate-limited por IP (hasheado) e audit-logado; a mesma action
  // alimenta a contagem do rate-limit.
  app.get('/api/certificados/verificar/:code', async (req: any, res) => {
    try {
      const code = String(req.params.code || '').toLowerCase();
      const ipHashed = hashClientIp(req) ?? '';
      const allowed = await checkRateLimit('', ipHashed, 'certificate_verify', 15, 30);
      if (!allowed) {
        return res.status(429).json({ error: 'Muitas tentativas. Aguarde alguns minutos.' });
      }
      await auditLog({
        query, action: 'certificate_verify',
        meta: { code_prefix: code.slice(0, 8) },
        ip: ipHashed,
      });

      // Formato inválido nem toca o banco. Resposta idêntica à de código
      // inexistente — não vazar distinção.
      if (!/^[0-9a-f]{32}$/.test(code)) return res.json({ valido: false });

      const { rows: [cert] } = await query(`
        SELECT ec.issued_at, ec.valid_until, ei.seal, ei.percentual,
               c.label AS cycle_label,
               COALESCE(p.trade_name, p.name) AS provider_name,
               pl.label AS location_label
          FROM eval_certificates ec
          JOIN eval_instances ei ON ei.id = ec.instance_id
          JOIN providers p ON p.id = ei.provider_id
          LEFT JOIN provider_locations pl ON pl.id = ei.location_id
          LEFT JOIN eval_cycles c ON c.id = ei.cycle_id
         WHERE ec.verify_code = $1
      `, [code]);
      if (!cert) return res.json({ valido: false });

      const expirado = new Date(cert.valid_until) < new Date();
      res.json({
        valido: true,
        expirado,
        seal: cert.seal,
        percentual: Number(cert.percentual) || 0,
        provider_name: cert.provider_name,
        location_label: cert.location_label,
        cycle_label: cert.cycle_label,
        issued_at: cert.issued_at,
        valid_until: cert.valid_until,
      });
    } catch (err: any) { console.error(err); res.status(500).json({ error: err.message }); }
  });
```

```tsx
// src/pages/VerificarCertificado.tsx
// Alvo do QR code (spec 5.4.5). Pública — sem login, como /politica-privacidade.
import React, { useEffect, useState } from 'react';
import { useParams } from 'react-router-dom';
import { BadgeCheck, BadgeX, Clock } from 'lucide-react';
import { fetchApi } from '../lib/api';

interface VerifyResult {
  valido: boolean;
  expirado?: boolean;
  seal?: string;
  percentual?: number;
  provider_name?: string;
  location_label?: string | null;
  cycle_label?: string | null;
  issued_at?: string;
  valid_until?: string;
}

const fmtDateOnly = (d: string) => {
  const [y, m, day] = String(d).slice(0, 10).split('-');
  return `${day}/${m}/${y}`;
};

export default function VerificarCertificado() {
  const { codigo } = useParams();
  const [result, setResult] = useState<VerifyResult | null>(null);
  const [error, setError] = useState('');

  useEffect(() => {
    fetchApi(`/certificados/verificar/${encodeURIComponent(String(codigo ?? ''))}`)
      .then(setResult)
      .catch((e: any) => setError(e.message || 'Erro ao verificar.'));
  }, [codigo]);

  return (
    <div className="grid min-h-screen place-items-center bg-gray-50 p-4">
      <div className="w-full max-w-md rounded-2xl border border-gray-200 bg-white p-8 shadow-sm">
        <h1 className="text-center text-lg font-bold text-secondary-900">
          Verificação de Certificado
        </h1>
        <p className="mt-1 text-center text-xs text-gray-500">
          Qualificação da Rede Credenciada — Unimed Criciúma
        </p>

        <div className="mt-6">
          {error ? (
            <p className="text-center text-sm text-red-600">{error}</p>
          ) : !result ? (
            <div className="flex justify-center py-8">
              <div className="h-8 w-8 animate-spin rounded-full border-b-2 border-emerald-600" />
            </div>
          ) : !result.valido ? (
            <div className="text-center">
              <BadgeX className="mx-auto h-12 w-12 text-red-500" />
              <p className="mt-2 font-bold text-red-600">Certificado não encontrado</p>
              <p className="mt-1 text-sm text-gray-500">
                O código informado não corresponde a nenhum certificado emitido.
              </p>
            </div>
          ) : (
            <div>
              <div className="text-center">
                {result.expirado ? (
                  <>
                    <Clock className="mx-auto h-12 w-12 text-amber-500" />
                    <p className="mt-2 font-bold text-amber-600">
                      Certificado expirado em {fmtDateOnly(result.valid_until!)}
                    </p>
                  </>
                ) : (
                  <>
                    <BadgeCheck className="mx-auto h-12 w-12 text-emerald-500" />
                    <p className="mt-2 font-bold text-emerald-600">Certificado válido</p>
                  </>
                )}
              </div>
              <dl className="mt-6 space-y-2 border-t pt-4 text-sm">
                {([
                  ['Prestador', result.provider_name],
                  ['Unidade', result.location_label],
                  ['Selo', result.seal],
                  ['Pontuação', result.percentual != null ? `${result.percentual.toFixed(1)}%` : null],
                  ['Ciclo', result.cycle_label],
                  ['Emitido em', result.issued_at ? new Date(result.issued_at).toLocaleDateString('pt-BR') : null],
                  ['Válido até', result.valid_until ? fmtDateOnly(result.valid_until) : null],
                ] as [string, string | null | undefined][]).map(([label, value]) => value ? (
                  <div key={label} className="flex justify-between gap-4">
                    <dt className="text-gray-500">{label}</dt>
                    <dd className="font-semibold text-secondary-900">{value}</dd>
                  </div>
                ) : null)}
              </dl>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```
