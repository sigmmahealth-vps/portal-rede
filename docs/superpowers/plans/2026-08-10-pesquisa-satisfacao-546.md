# Pesquisa de Satisfação nativa (5.4.6) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Formulário de Pesquisa de Satisfação nativo, com prazo de 7 dias, bônus automático de 10 pontos no selo e Dashboard de Indicadores (NPS/CSAT) para o staff.

**Architecture:** Uma tabela `eval_survey_responses` (1:1 com `eval_instances`), endpoints do prestador (`/me/survey`) e do staff (`/eval/surveys`), integração no `/submit` (grava `submitted_at`) e no `/parecer` (bloqueio + bônus). Duas funções puras testáveis (`surveyStats`, `surveyBonus`) concentram a lógica. UI: aba do prestador em `PrestadorView` e aba de dashboard em `QualificacaoStaff`.

**Tech Stack:** Node/Express (server.ts), Postgres (migrations SQL), React + Tailwind, lucide-react, testes com `node:test` + `tsx`.

## Global Constraints

- Prazo do bônus: **7 dias corridos** após o envio do roteiro (`eval_instances.submitted_at`).
- Bônus: **+10 pontos percentuais, teto 100**, só quando respondida no prazo; senão 0.
- Selo: `computeSeal(percentualFinal, elim)` — faixas Ouro >90, Prata >60, Bronze ≤60.
- NPS de Q6 (0–10): promotores 9–10, passivos 7–8, detratores 0–6; `NPS = %promotores − %detratores`.
- Q3 opções: `sim_totalmente | sim_parcialmente | neutro | nao_precisei_ajuda | nao_insuficientes` (`neutro` = "Indiferente — não ajudaram nem atrapalharam").
- Anti-IDOR: rotas do prestador sempre escopadas por `req.user.provider_id`, nunca pelo body.
- Rodar testes: `node --import tsx --test <arquivo>`. Typecheck: `node node_modules/typescript/bin/tsc --noEmit`.
- Node fora do PATH nesta máquina: prefixar `export PATH="/c/Program Files/nodejs:$PATH"` (bash) ou usar caminho absoluto do node.

```ts
// Prazo e bônus da Pesquisa de Satisfação (spec 5.4.6).
export const SURVEY_DEADLINE_DAYS = 7;

/** Fim do prazo = envio do roteiro + 7 dias corridos. */
export function surveyDeadline(roteiroSubmittedAt: Date): Date {
  const d = new Date(roteiroSubmittedAt.getTime());
  d.setUTCDate(d.getUTCDate() + SURVEY_DEADLINE_DAYS);
  return d;
}

/** True se `at` está dentro do prazo (inclusive a borda exata dos 7 dias). */
export function isWithinDeadline(roteiroSubmittedAt: Date, at: Date): boolean {
  return at.getTime() <= surveyDeadline(roteiroSubmittedAt).getTime();
}

/** Bônus percentual aplicado ao selo. */
export function bonusPoints(withinDeadline: boolean): number {
  return withinDeadline ? 10 : 0;
}
```

```ts
// Indicadores da Pesquisa de Satisfação (spec 5.4.6): NPS, CSAT e distribuições.
export interface SurveyRow {
  q1_clareza: number | null;
  q2_portal: number | null;
  q3_documentos: string | null;
  q4_dificuldade: boolean | null;
  q5_geral: number | null;
  q6_nps: number | null;
}

export interface SurveyStats {
  count: number;
  nps: { promoters: number; passives: number; detractors: number; score: number | null };
  csat: { q1: number | null; q2: number | null; q5: number | null };
  q3: Record<string, number>;
  q4DifficultyPct: number | null;
}

const round1 = (n: number): number => Math.round(n * 10) / 10;

function mean(values: (number | null)[]): number | null {
  const nums = values.filter((v): v is number => v != null);
  if (!nums.length) return null;
  return round1(nums.reduce((a, b) => a + b, 0) / nums.length);
}

export function computeSurveyStats(rows: SurveyRow[]): SurveyStats {
  const nps = rows.map(r => r.q6_nps).filter((v): v is number => v != null);
  const promoters = nps.filter(v => v >= 9).length;
  const detractors = nps.filter(v => v <= 6).length;
  const passives = nps.length - promoters - detractors;
  const score = nps.length
    ? Math.round(((promoters - detractors) / nps.length) * 100)
    : null;

  const q3: Record<string, number> = {};
  for (const r of rows) if (r.q3_documentos) q3[r.q3_documentos] = (q3[r.q3_documentos] ?? 0) + 1;

  const q4 = rows.map(r => r.q4_dificuldade).filter((v): v is boolean => v != null);
  const q4DifficultyPct = q4.length
    ? round1((q4.filter(Boolean).length / q4.length) * 100)
    : null;

  return {
    count: rows.length,
    nps: { promoters, passives, detractors, score },
    csat: {
      q1: mean(rows.map(r => r.q1_clareza)),
      q2: mean(rows.map(r => r.q2_portal)),
      q5: mean(rows.map(r => r.q5_geral)),
    },
    q3,
    q4DifficultyPct,
  };
}
```

```tsx
import { Star, Send, Loader2, Smile, CheckCircle } from 'lucide-react';

const Q3_OPTS = [
  { v: 'sim_totalmente', label: 'Sim, totalmente' },
  { v: 'sim_parcialmente', label: 'Sim, parcialmente' },
  { v: 'neutro', label: 'Indiferente — não ajudaram nem atrapalharam' },
  { v: 'nao_precisei_ajuda', label: 'Não, precisei de ajuda' },
  { v: 'nao_insuficientes', label: 'Não, foram insuficientes' },

function Stars({ value, onChange }: { value: number; onChange: (n: number) => void }) {
  return (
    <div className="flex gap-1">
      {[1, 2, 3, 4, 5].map(n => (
        <button key={n} type="button" onClick={() => onChange(n)} aria-label={`${n} estrelas`}
          className="transition-transform hover:scale-110">
          <Star className={`h-7 w-7 ${n <= value ? 'fill-amber-400 text-amber-400' : 'text-gray-300'}`} />
        </button>
      ))}
    </div>
  );

function fmtDate(iso: string): string {
  try { return new Date(iso).toLocaleDateString('pt-BR'); } catch { return iso; }

export default function PesquisaSatisfacao() {
  const [data, setData] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [form, setForm] = useState({
    q1_clareza: 0, q2_portal: 0, q3_documentos: '', q4_dificuldade: null as boolean | null,
    q4_texto: '', q5_geral: 0, q6_nps: null as number | null, comentarios: '',

    finally { setLoading(false); }
  };

  const submit = async () => {
    setSaving(true);
    try {
      await fetchApi('/me/survey', { method: 'POST', body: JSON.stringify(form) });
      await load();

    finally { setSaving(false); }

      <div className="rounded-2xl border border-dashed border-gray-300 bg-white py-20 text-center">
        <div className="mx-auto mb-4 grid h-14 w-14 place-items-center rounded-full bg-emerald-50 text-emerald-600"><Smile className="h-7 w-7" /></div>

      </div>
    );

  if (data?.state === 'respondida') {
    const r = data.response;
    return (
      <div className="rounded-2xl border border-emerald-200 bg-emerald-50 p-8 text-center">
        <CheckCircle className="mx-auto h-12 w-12 text-emerald-500" />
        <h3 className="mt-2 text-lg font-bold text-secondary-900">Pesquisa respondida</h3>
        <p className="mt-1 text-sm text-gray-600">Enviada em {fmtDate(r.submitted_at)}.</p>
        <p className="mt-1 text-sm font-semibold text-emerald-700">
          {r.bonus_points > 0 ? `+${r.bonus_points} pontos bônus aplicados à sua qualificação.` : 'Enviada após o prazo — sem pontos bônus.'}
        </p>
      </div>
    );
  }

  // state === 'disponivel'
  const expired = !!data?.expired;
  const canSubmit = form.q1_clareza > 0 && form.q2_portal > 0 && form.q3_documentos
    && form.q4_dificuldade !== null && form.q5_geral > 0 && form.q6_nps != null && !saving;

  return (
    <div className="space-y-6">
      <div className={`flex items-start gap-4 rounded-2xl border p-5 ${expired ? 'border-gray-200 bg-gray-50' : 'border-emerald-200 bg-emerald-50'}`}>
        <div className="grid h-10 w-10 shrink-0 place-items-center rounded-full bg-white text-amber-500"><Star className="h-6 w-6 fill-amber-400 text-amber-400" /></div>

            {expired ? 'Prazo encerrado — você ainda pode responder, sem pontos bônus.' : 'Responda a pesquisa e ganhe 10 pontos bônus!'}
          </p>
          <p className="mt-1 text-sm text-gray-600">Sua opinião é muito importante para melhorarmos o processo de qualificação. Leva menos de 2 minutos.</p>
          {!expired && <p className="mt-1 text-sm font-semibold text-emerald-700">Disponível até {fmtDate(data.deadline)} (7 dias após o envio do roteiro).</p>}
        </div>
      </div>

      <div className="space-y-6 rounded-2xl border border-gray-200 bg-white p-6">
        <div>
          <p className="font-semibold text-secondary-900">1. Como você avalia a clareza das instruções e orientações do roteiro de qualificação?</p>
          <div className="mt-2"><Stars value={form.q1_clareza} onChange={n => set({ q1_clareza: n })} /></div>
          <p className="mt-1 text-xs text-gray-400">1 = Muito ruim | 5 = Excelente</p>
        </div>
        <div>
          <p className="font-semibold text-secondary-900">2. Como você avalia a facilidade de uso do portal para envio das evidências?</p>
          <div className="mt-2"><Stars value={form.q2_portal} onChange={n => set({ q2_portal: n })} /></div>
          <p className="mt-1 text-xs text-gray-400">1 = Muito ruim | 5 = Excelente</p>
        </div>
        <div>
          <p className="font-semibold text-secondary-900">3. As orientações sobre quais documentos enviar foram suficientes e claras?</p>
          <div className="mt-2 space-y-1.5">
            {Q3_OPTS.map(o => (
              <label key={o.v} className="flex items-center gap-2 text-sm text-secondary-900">

              </label>
            ))}

              <button key={o.l} type="button" onClick={() => set({ q4_dificuldade: o.v })}
                className={`rounded-lg border px-4 py-1.5 text-sm font-semibold ${form.q4_dificuldade === o.v ? 'border-emerald-500 bg-emerald-50 text-emerald-700' : 'border-gray-300 text-gray-700 hover:bg-gray-50'}`}>

          </div>

              className="mt-2 w-full rounded-lg border border-gray-300 p-2 text-sm text-secondary-900" />

          <p className="font-semibold text-secondary-900">5. Como você avalia o processo de qualificação da Unimed Criciúma de forma geral?</p>

          <p className="mt-1 text-xs text-gray-400">1 = Muito ruim | 5 = Excelente</p>
        </div>
        <div>
          <p className="font-semibold text-secondary-900">6. De 0 a 10, o quanto você recomendaria o Portal da Rede Credenciada da Unimed Criciúma a outro prestador?</p>
```

```tsx
import React, { useEffect, useMemo, useState } from 'react';
import { fetchApi } from '../../lib/api';
import { computeSurveyStats, SurveyRow } from '../../lib/surveyStats';
import { Loader2, MessageSquare } from 'lucide-react';

const Q3_LABEL: Record<string, string> = {
  sim_totalmente: 'Sim, totalmente', sim_parcialmente: 'Sim, parcialmente',
  neutro: 'Indiferente', nao_precisei_ajuda: 'Não, precisei de ajuda',
  nao_insuficientes: 'Não, foram insuficientes',
};

function fmtDate(iso: string): string {
  try { return new Date(iso).toLocaleDateString('pt-BR'); } catch { return iso; }
}

function Card({ label, value, sub }: { label: string; value: React.ReactNode; sub?: string }) {
  return (
    <div className="rounded-xl border border-gray-100 bg-white p-4 shadow-sm">
      <p className="text-[11px] font-semibold uppercase tracking-wide text-gray-400">{label}</p>
      <p className="mt-1 text-2xl font-bold text-secondary-900">{value}</p>
      {sub && <p className="text-xs text-gray-500">{sub}</p>}
    </div>
  );
}

export default function PesquisaDashboard({ user }: { user: any }) {
  const [cycles, setCycles] = useState<any[]>([]);
  const [cycleId, setCycleId] = useState<number | null>(null);
  const [rows, setRows] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => { (async () => {
    const cs = await fetchApi('/eval/cycles');
    setCycles(cs);
    const active = cs.find((c: any) => c.active) || cs[0];
    if (active) setCycleId(active.id);
  })(); }, []);

  useEffect(() => {
    if (cycleId == null) return;
    setLoading(true);
    fetchApi(`/eval/surveys?cycle_id=${cycleId}`).then(setRows).finally(() => setLoading(false));
  }, [cycleId]);

  const stats = useMemo(() => computeSurveyStats(rows as SurveyRow[]), [rows]);

  return (
    <div className="space-y-6">
      <div className="flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
        <div>
          <h1 className="text-2xl font-bold text-gray-900">Pesquisa de Satisfação</h1>
          <p className="text-sm text-gray-500">Indicadores da rede credenciada por ciclo</p>
        </div>
        <select className="rounded-lg border border-gray-300 bg-white px-3 py-2 text-sm"
          value={cycleId ?? ''} onChange={e => setCycleId(Number(e.target.value))}>
          {cycles.map((c: any) => <option key={c.id} value={c.id}>{c.label}{c.active ? ' (ativo)' : ''}</option>)}
        </select>
      </div>

      {loading ? (
        <div className="flex justify-center py-20"><Loader2 className="h-8 w-8 animate-spin text-emerald-600" /></div>
      ) : rows.length === 0 ? (
        <p className="rounded-xl border border-gray-100 bg-white p-8 text-center text-gray-500">Nenhuma resposta neste ciclo.</p>
      ) : (
        <>
          <div className="grid grid-cols-2 gap-4 md:grid-cols-4">
            <Card label="NPS" value={stats.nps.score ?? '—'} sub={`${stats.nps.promoters} prom · ${stats.nps.passives} pass · ${stats.nps.detractors} detr`} />
            <Card label="Respostas" value={stats.count} />
            <Card label="CSAT — Clareza (Q1)" value={stats.csat.q1 ?? '—'} sub="média 1–5" />
            <Card label="CSAT — Portal (Q2)" value={stats.csat.q2 ?? '—'} sub="média 1–5" />
            <Card label="CSAT — Processo (Q5)" value={stats.csat.q5 ?? '—'} sub="média 1–5" />
            <Card label="Dificuldade técnica (Q4)" value={stats.q4DifficultyPct != null ? `${stats.q4DifficultyPct}%` : '—'} sub="responderam Sim" />
            <Card label="Documentos claros (Q3)" value={stats.q3['sim_totalmente'] ?? 0} sub="Sim, totalmente" />
          </div>

          <div className="overflow-x-auto rounded-xl border border-gray-100 bg-white shadow-sm">
            <table className="w-full text-left text-sm">
              <thead className="bg-gray-50 text-xs uppercase tracking-wide text-gray-500">
                <tr>
                  <th className="p-3">Prestador</th><th className="p-3">Enviada</th>
                  <th className="p-3">NPS</th><th className="p-3">Q1</th><th className="p-3">Q2</th>
                  <th className="p-3">Q3</th><th className="p-3">Q5</th><th className="p-3">Dificuldade</th>
                  <th className="p-3">Comentários</th>
                </tr>
              </thead>
              <tbody>
                {rows.map((r: any) => (
                  <tr key={r.id} className="border-t border-gray-50 align-top">
                    <td className="p-3 font-medium text-gray-900">{r.provider_name}</td>
                    <td className="p-3 text-gray-600">{fmtDate(r.submitted_at)}</td>
                    <td className="p-3 font-bold text-gray-700">{r.q6_nps ?? '—'}</td>
                    <td className="p-3 text-gray-600">{r.q1_clareza ?? '—'}</td>
                    <td className="p-3 text-gray-600">{r.q2_portal ?? '—'}</td>
                    <td className="p-3 text-gray-600">{Q3_LABEL[r.q3_documentos] || '—'}</td>
                    <td className="p-3 text-gray-600">{r.q5_geral ?? '—'}</td>
                    <td className="p-3 text-gray-600">
                      {r.q4_dificuldade ? 'Sim' : 'Não'}{r.q4_texto ? ` — ${r.q4_texto}` : ''}
                    </td>
                    <td className="p-3 text-gray-600">
                      {r.comentarios ? <span className="flex gap-1"><MessageSquare className="mt-0.5 h-3 w-3 shrink-0 text-gray-400" />{r.comentarios}</span> : '—'}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </>
      )}
    </div>
  );
}
```
