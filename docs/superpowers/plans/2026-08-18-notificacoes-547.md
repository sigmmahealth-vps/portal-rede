# Notificações Automáticas 5.4.7 (S5) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Notificar prestador e auditoria (portal + e-mail) nos 12 eventos da spec 5.4.7, com sininho real, página "Ver todas" e provisionamento manual de acesso do prestador com convite.

**Architecture:** Tabela única `notifications` com outbox embutido (status do e-mail na própria linha; retry pelo job horário existente). Catálogo de textos puro em `src/lib/notifyCatalog.ts`; emissor `notifyEvent()` + resolução de destinatários no `server.ts`; eventos transacionais emitidos inline nos endpoints existentes; eventos de prazo varridos pelo `setInterval` horário. Spec: `docs/superpowers/specs/2026-08-18-notificacoes-547-design.md`.

**Tech Stack:** Node/Express + pg (server.ts monolítico), Resend (e-mail texto puro), React + react-router (frontend), node:test para testes.

-- 0030_notifications.sql
-- Notificações (spec 5.4.7). Uma linha por destinatário; a própria tabela é o
-- outbox de e-mail (email_status/email_attempts, retry no job horário).
-- user_id NULL = canal e-mail-apenas (ex.: roteiro_pendente); email_to NULL = só portal.

CREATE TABLE IF NOT EXISTS notifications (
  id             bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id        bigint REFERENCES users(id) ON DELETE CASCADE,
  email_to       text,
  provider_id    bigint REFERENCES providers(id) ON DELETE CASCADE,
  type           text NOT NULL,
  title          text NOT NULL,
  body           text,
  link           text,
  dedupe_key     text UNIQUE,
  read_at        timestamptz,
  email_status   text NOT NULL DEFAULT 'na'
                   CHECK (email_status IN ('na','pendente','enviado','falhou')),
  email_attempts int NOT NULL DEFAULT 0,
  created_at     timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS notifications_user_read_idx
  ON notifications (user_id, read_at) WHERE user_id IS NOT NULL;
CREATE INDEX IF NOT EXISTS notifications_email_retry_idx
  ON notifications (email_status) WHERE email_status IN ('pendente','falhou');
```

// src/lib/notifyCatalog.ts
// Catálogo de textos das notificações (spec 5.4.7). Puro: sem DB, sem env.
// O link é relativo; o server prefixa APP_URL no corpo do e-mail.

export const NOTIFY_TYPES = [
  'ciclo_aberto', 'roteiro_enviado', 'reanalise_solicitada',
  'parecer_aprovado', 'parecer_reprovado', 'certificado_emitido',
  'plano_gerado', 'acao_aguardando_validacao',
  'acao_vencendo', 'acao_atrasada', 'roteiro_pendente', 'certificado_vencendo',
] as const;
export type NotifyType = typeof NOTIFY_TYPES[number];

export interface NotifyContent { title: string; body: string; link: string; }

// Datas date-only ('YYYY-MM-DD') ganham T12:00 para não voltar um dia por fuso.
function fmtDate(v: string | Date | null | undefined): string {
  if (!v) return '';
  const d = typeof v === 'string' ? new Date(v.length === 10 ? `${v}T12:00:00` : v) : v;
  return isNaN(d.getTime()) ? '' : d.toLocaleDateString('pt-BR');
}

export function dedupeKey(type: string, ref: string, dest: string | number): string {
  return `${type}:${ref}:${dest}`;
}

export function renderNotification(type: NotifyType, ctx: Record<string, any> = {}): NotifyContent {
  const link = '/qualificacao';
  switch (type) {
    case 'ciclo_aberto':
      return { title: `Ciclo de qualificação ${ctx.cycle_label ?? ''} aberto`.replace(/\s+/g, ' ').trim(),
               body: 'Seu roteiro de qualificação está disponível para preenchimento no portal.', link };
    case 'roteiro_enviado':
      return { title: 'Roteiro enviado para análise',
               body: `${ctx.provider_name ?? 'Um prestador'} enviou o roteiro de qualificação para análise.`, link };
    case 'reanalise_solicitada':
      return ctx.integral
        ? { title: 'Avaliação devolvida para reanálise',
            body: 'O auditor devolveu sua avaliação. Revise os critérios apontados e reenvie.', link }
        : { title: `Reanálise solicitada — critério ${ctx.question_code ?? ''}`.trim(),
            body: 'O auditor solicitou reanálise de um critério da sua avaliação. Ajuste a evidência e reenvie.', link };
    case 'parecer_aprovado':
      return { title: `Avaliação aprovada — selo ${ctx.seal ?? ''}`.trim(),
               body: `Sua avaliação de qualificação foi aprovada com ${Math.round(Number(ctx.percentual ?? 0))}% de atingimento. Veja o parecer no portal.`, link };
    case 'parecer_reprovado':
      return { title: 'Avaliação concluída — não qualificado',
               body: 'Sua avaliação foi concluída sem atingir os requisitos do selo. Veja o parecer e o plano de ação no portal.', link };
    case 'certificado_emitido':
      return { title: 'Certificado de qualificação emitido',
               body: `Seu certificado está disponível no portal. Validade: ${fmtDate(ctx.valid_until)}.`, link };
    case 'plano_gerado':
      return { title: `Plano de ação gerado${ctx.provider_name ? ` — ${ctx.provider_name}` : ''}`,
               body: ctx.items
                 ? `Um plano de ação com ${ctx.items} ação(ões) foi gerado. Acompanhe os prazos no portal.`
                 : 'Um plano de ação foi criado para acompanhamento. Acesse o portal.', link };
    case 'acao_aguardando_validacao':
      return { title: `Ação concluída aguardando validação${ctx.provider_name ? ` — ${ctx.provider_name}` : ''}`,
               body: `O prestador concluiu a ação "${ctx.what ?? ''}" e aguarda validação do auditor.`, link };
    case 'acao_vencendo':
      return { title: 'Ação do plano próxima do vencimento',
               body: `A ação "${ctx.what ?? ''}" vence em ${fmtDate(ctx.due_date)}.`, link };
    case 'acao_atrasada':
      return { title: `Ação do plano em atraso${ctx.provider_name ? ` — ${ctx.provider_name}` : ''}`,
               body: `A ação "${ctx.what ?? ''}" venceu em ${fmtDate(ctx.due_date)} e não foi concluída.`, link };
    case 'roteiro_pendente':
      return ctx.digest
        ? { title: `${ctx.count ?? 0} roteiro(s) pendente(s) — ciclo ${ctx.cycle_label ?? ''} encerra em ${fmtDate(ctx.end_date)}`,
            body: 'Há prestadores que ainda não enviaram o roteiro de qualificação. Veja o painel do ciclo.', link }
        : { title: `Roteiro pendente — o ciclo encerra em ${fmtDate(ctx.end_date)}`,
            body: 'Você ainda não enviou seu roteiro de qualificação. O prazo termina junto com o ciclo.', link };
    case 'certificado_vencendo':
      return { title: 'Certificado próximo do vencimento',
               body: `Seu certificado de qualificação vence em ${fmtDate(ctx.valid_until)}.`, link };
  }
}
```

```ts
  // ===== Notificações (spec 5.4.7) =====
  type NotifyRecipient = { user_id: number | null; email: string | null };

  async function prestadorRecipients(providerId: number): Promise<NotifyRecipient[]> {
    const { rows } = await query(
      `SELECT id AS user_id, email FROM users
        WHERE provider_id = $1 AND is_active = true AND role IN ('prestador','cooperado')`,
      [providerId]);
    return rows.map(r => ({ user_id: Number(r.user_id), email: r.email }));
  }

  async function auditoriaRecipients(): Promise<NotifyRecipient[]> {
    const { rows } = await query(
      `SELECT id AS user_id, email FROM users WHERE role = 'auditoria' AND is_active = true`);
    return rows.map(r => ({ user_id: Number(r.user_id), email: r.email }));
  }

  // Envia o e-mail de uma notification e atualiza o outbox. O UPDATE inicial é o
  // "claim": incrementa attempts atomicamente e some da fila se outro tick pegou.
  async function sendNotificationEmail(rowId: number, to: string, title: string, body: string, link: string) {
    try {
      const { rows: claimed } = await query(
        `UPDATE notifications SET email_attempts = email_attempts + 1
          WHERE id = $1 AND email_attempts < 3 AND email_status IN ('pendente','falhou')
          RETURNING id`, [rowId]);
      if (!claimed.length) return;
      const appUrl = process.env.APP_URL ?? `http://localhost:${PORT}`;
      await mailer.sendNotification(to, `Portal Unimed — ${title}`, `${body}\n\nAcesse: ${appUrl}${link}`);
      await query(`UPDATE notifications SET email_status = 'enviado' WHERE id = $1`, [rowId]);
    } catch (e) {
      console.error(`[notify] e-mail falhou (notification ${rowId}):`, e);
      await query(`UPDATE notifications SET email_status = 'falhou' WHERE id = $1`, [rowId]).catch(() => {});
    }
  }

  // Grava as notificações e dispara os e-mails (fire-and-forget). NUNCA lança.
  // opts.email=false → só portal; opts.portal=false → só e-mail;
  // opts.dedupeRef → eventos de prazo (disparo único por item×destinatário).
  async function notifyEvent(
    type: NotifyType,
    ctx: Record<string, any>,
    recipients: NotifyRecipient[],
    opts: { providerId?: number; email?: boolean; portal?: boolean; dedupeRef?: string } = {},
  ) {
    const wantEmail = opts.email !== false;
    const wantPortal = opts.portal !== false;
    try {
      const c = renderNotification(type, ctx);
      for (const r of recipients) {
        const userId = wantPortal ? r.user_id : null;
        const emailTo = wantEmail ? r.email : null;
        if (!userId && !emailTo) continue;
        const key = opts.dedupeRef ? dedupeKey(type, opts.dedupeRef, r.user_id ?? r.email ?? '') : null;
        const { rows } = await query(`
          INSERT INTO notifications (user_id, email_to, provider_id, type, title, body, link, dedupe_key, email_status)
          VALUES ($1, $2, $3, $4, $5, $6, $7, $8, CASE WHEN $2::text IS NULL THEN 'na' ELSE 'pendente' END)
          ON CONFLICT (dedupe_key) DO NOTHING
          RETURNING id`,
          [userId, emailTo, opts.providerId ?? null, type, c.title, c.body, c.link, key]);
        const row = rows[0];
        if (row && emailTo) void sendNotificationEmail(Number(row.id), emailTo, c.title, c.body, c.link);
      }
    } catch (e) {
      console.error(`[notify] falha ao emitir ${type}:`, e);
    }
  }
```

```ts
  // Job de notificações (spec 5.4.7): varre condições de prazo (disparo único via
  // dedupe_key) e re-tenta e-mails pendentes/falhos. Idempotente por construção.
  async function runNotificationJobs() {
    try {
      // 1) Ação a vencer em até 15 dias — prestador.
      const { rows: vencendo } = await query(`
        SELECT it.id, it.what, it.due_date, ap.provider_id
          FROM eval_action_items it
          JOIN eval_action_plans ap ON ap.id = it.plan_id
         WHERE it.status <> 'concluida'
           AND COALESCE(it.closed_without_completion, false) = false
           AND it.due_date BETWEEN CURRENT_DATE AND CURRENT_DATE + 15`);
      for (const it of vencendo) {
        await notifyEvent('acao_vencendo', { what: it.what, due_date: it.due_date },
          await prestadorRecipients(Number(it.provider_id)),
          { providerId: Number(it.provider_id), dedupeRef: `item:${it.id}` });
      }

      // 2) Ação em atraso — prestador + auditoria.
      const { rows: atrasadas } = await query(`
        SELECT it.id, it.what, it.due_date, ap.provider_id, p.name AS provider_name
          FROM eval_action_items it
          JOIN eval_action_plans ap ON ap.id = it.plan_id
          JOIN providers p ON p.id = ap.provider_id
         WHERE it.status <> 'concluida'
           AND COALESCE(it.closed_without_completion, false) = false
           AND it.due_date < CURRENT_DATE`);
      for (const it of atrasadas) {
        await notifyEvent('acao_atrasada',
          { what: it.what, due_date: it.due_date, provider_name: it.provider_name },
          [...await prestadorRecipients(Number(it.provider_id)), ...await auditoriaRecipients()],
          { providerId: Number(it.provider_id), dedupeRef: `item:${it.id}` });
      }

      // 3) Roteiro pendente a 30 dias do fim do ciclo — SÓ E-MAIL (spec).
      //    Prestador: individual por instância. Auditoria: resumo por ciclo.
      const { rows: pendentes } = await query(`
        SELECT ei.id, ei.provider_id, c.id AS cycle_id, c.label, c.end_date
          FROM eval_instances ei
          JOIN eval_cycles c ON c.id = ei.cycle_id
         WHERE ei.status IN ('pendente', 'em_andamento')
           AND c.active = true
           AND c.end_date - 30 <= CURRENT_DATE AND c.end_date >= CURRENT_DATE`);
      for (const inst of pendentes) {
        await notifyEvent('roteiro_pendente', { end_date: inst.end_date },
          await prestadorRecipients(Number(inst.provider_id)),
          { providerId: Number(inst.provider_id), dedupeRef: `inst:${inst.id}`, portal: false });
      }
      const byCycle = new Map<number, { label: string; end_date: string; count: number }>();
      for (const inst of pendentes) {
        const c = byCycle.get(Number(inst.cycle_id)) ?? { label: inst.label, end_date: inst.end_date, count: 0 };
        c.count++; byCycle.set(Number(inst.cycle_id), c);
      }
      for (const [cycleId, c] of byCycle) {
        await notifyEvent('roteiro_pendente',
          { digest: true, count: c.count, cycle_label: c.label, end_date: c.end_date },
          await auditoriaRecipients(), { dedupeRef: `cycle:${cycleId}`, portal: false });
      }

      // 4) Certificado a vencer em até 30 dias — prestador.
      const { rows: certs } = await query(`
        SELECT ec.id, ec.valid_until, ei.provider_id
          FROM eval_certificates ec
          JOIN eval_instances ei ON ei.id = ec.instance_id
         WHERE ec.valid_until - 30 <= CURRENT_DATE AND ec.valid_until >= CURRENT_DATE`);
      for (const ct of certs) {
        await notifyEvent('certificado_vencendo', { valid_until: ct.valid_until },
          await prestadorRecipients(Number(ct.provider_id)),
          { providerId: Number(ct.provider_id), dedupeRef: `cert:${ct.id}` });
      }

      // 5) Retry do outbox de e-mail (o claim em sendNotificationEmail evita corrida).
      const { rows: retry } = await query(`
        SELECT id, email_to, title, body, link FROM notifications
         WHERE email_status IN ('pendente','falhou') AND email_attempts < 3 AND email_to IS NOT NULL
         ORDER BY id LIMIT 50`);
      for (const n of retry) {
        await sendNotificationEmail(Number(n.id), n.email_to, n.title, n.body ?? '', n.link ?? '/qualificacao');
      }
    } catch (e) {
      console.error('[notify] job de notificações falhou:', e);
    }
  }
```

import { useEffect, useRef, useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { Bell } from 'lucide-react';

interface Notif { id: number; title: string; body: string | null; link: string | null; read_at: string | null; created_at: string; }

export default function NotificationsBell() {
  const [open, setOpen] = useState(false);
  const [count, setCount] = useState(0);
  const [items, setItems] = useState<Notif[]>([]);
  const ref = useRef<HTMLDivElement>(null);
  const navigate = useNavigate();

  const refreshCount = () => {
    if (document.visibilityState !== 'visible') return;
    fetchApi('/notifications/unread-count').then(d => setCount(d.count)).catch(() => {});
  };

  useEffect(() => {
    refreshCount();
    const t = setInterval(refreshCount, 60_000);
    return () => clearInterval(t);
  }, []);

  useEffect(() => {
    const onClick = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener('mousedown', onClick);
    return () => document.removeEventListener('mousedown', onClick);

    const next = !open;
    setOpen(next);
    if (next) {
      try { setItems((await fetchApi('/notifications')).items); } catch { /* dropdown vazio */ }
    }
  };

  const openItem = async (n: Notif) => {
    setOpen(false);
    if (!n.read_at) {
      fetchApi(`/notifications/${n.id}/read`, { method: 'POST' }).catch(() => {});
      setCount(c => Math.max(0, c - 1));
    }
    navigate(n.link || '/qualificacao');
  };

  const markAll = async () => {
    try {
      await fetchApi('/notifications/read-all', { method: 'POST' });
      setCount(0);
      setItems(prev => prev.map(n => ({ ...n, read_at: n.read_at ?? new Date().toISOString() })));
    } catch { /* mantém como está */ }
  };

  return (

      <button onClick={toggle} className="text-gray-500 hover:text-[#00995D] transition-colors relative" title="Notificações">

          <span className="absolute -top-1.5 -right-1.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-red-500 px-1 text-[10px] font-bold text-white ring-2 ring-[#f4f4f4]">
            {count > 99 ? '99+' : count}
          </span>
        )}
      </button>
      {open && (
        <div className="absolute right-0 mt-2 w-80 bg-white rounded-xl shadow-lg border border-gray-100 py-2 z-50">
          <div className="flex items-center justify-between px-4 py-2 border-b border-gray-100">
            <h3 className="font-bold text-gray-800">Notificações</h3>
            {count > 0 && (
              <button onClick={markAll} className="text-xs font-medium text-[#00995D] hover:underline">
                Marcar todas como lidas
              </button>
            )}
          </div>
          <div className="max-h-64 overflow-y-auto">
            {items.length === 0 && <p className="px-4 py-6 text-center text-sm text-gray-400">Nenhuma notificação.</p>}
